// --- FILTERED SEARCHES ---

// Query: Salary > 4000
db.instructors.find({ salary: { $gt : 4000 } }, { firstName : 1, salary : 1 });
/*
Responses:
{ _id: 8, firstName: 'mazen', salary: 7040 }
{ _id: 9, firstName: 'ebtesam', salary: 7500 }
*/

// Query: Age <= 25
db.instructors.find({ age: { $lte : 25 } });
// Response: (Returns all 4 instructors with IDs 6, 7, 8, and 9)

// Query: Nested address search (City: Mansoura, Street: 10 or 14)
db.instructors.find({ "address.city" : "mansoura", "address.street" : { $in : [10, 14] } });
/*
Response:
{ _id: 9, firstName: 'ebtesam', lastName: 'hesham', ..., address: { city: 'mansoura', street: 14 ... } }
*/

// --- ITERATION AND ARRAY HANDLING ---

// ATTEMPT 1: (Failed due to missing 'courses' field in some docs)
db.instructors.find({}, { firstName: 1, courses: 1 }).forEach(rec => {
    print(`courses number ${rec.courses.length}`)
});
// Response: TypeError: Cannot read properties of undefined (reading 'length')

// ATTEMPT 2: (Success using Optional Chaining)
db.instructors.find({}, { firstName: 1, courses: 1 }).forEach(rec => {
    print(`courses number ${rec.courses?.length}`)
});
/*
Responses:
courses number undefined (Doc 1)

courses number 4         (Doc 6)

// --- SORTING ---
db.instructors.find({}, { firstName: 1, lastName: 1, age: 1 })
    .sort({ firstName: 1, lastName: -1 })
    .forEach(rec => {
        print(`FullName : ${rec.firstName} ${rec.lastName} , Age : ${rec.age}`)
    });
/*
Responses:

*/

// --- UPDATES AND DELETIONS ---

// Delete specific user
db.instructors.deleteOne({ firstName: "ebtesam", courses: { $size: 5 } });
// Response: { acknowledged: true, deletedCount: 1 }

// Set all as active
db.instructors.updateMany({}, { $set: { active: true } });
// Response: { matchedCount: 5, modifiedCount: 5 }

// Update specific array element (EF -> jquery)
db.instructors.updateOne(
    { firstName: "mazen", lastName: "mohammed", courses: "EF" }, 
    { $set: { "courses.$": "jquery" } }
);
// Response: { matchedCount: 1, modifiedCount: 1 }

// Increment/Decrement Salary
db.instructors.updateMany({ courses: { $size: 3 } }, { $inc: { salary: -500 } });
// Response: { matchedCount: 1, modifiedCount: 1 }

// Rename field: address -> fullAddress
db.instructors.updateMany({}, { $rename: { address: "fullAddress" } });
// Response: { matchedCount: 5, modifiedCount: 3 }

// Update nested field after rename
db.instructors.updateOne({ firstName: "noha" }, { $set: { "fullAddress.street": 20 } });
// Response: { matchedCount: 1, modifiedCount: 1 }
