  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["facultyName", "address"],
      properties: {
        facultyName: { bsonType: "string" },
        address: { bsonType: "string" }
      }
    }
  }
});

db.createCollection("course", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["courseName", "finalMark"],
      properties: {
        courseName: { bsonType: "string" },
        finalMark: { bsonType: "int", minimum: 0, maximum: 100 }
      }
    }
  }
});

db.course.insertMany([
  { _id: "JS101", courseName: "JavaScript Basics", finalMark: 100 },
  { _id: "DB202", courseName: "MongoDB Advanced", finalMark: 100 },
  { _id: "SEA303", courseName: "Marine Navigation Systems", finalMark: 100 }
]);

db.createCollection("student", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["firstName", "lastName", "isFired", "facultyID", "results"],
      properties: {
        firstName: { bsonType: "string" },
        lastName: { bsonType: "string" },
        isFired: { bsonType: "bool" },
        facultyID: { bsonType: "int" },
        results: {
          bsonType: "array",
          items: {
            bsonType: "object",
            required: ["courseID", "grade"],
            properties: {
              courseID: { bsonType: "string" },
              grade: { bsonType: "int", minimum: 0, maximum: 100 }
            }
          }
        }
      }
    }
  }
});

2- db.student.aggregate([
  {
    $project: {
      _id: 0,
      fullName: { 
        $concat: ["$firstName", " ", "$lastName"]

      averageGrade: { 
        $avg: "$results.grade"

    }
  }
])

3- db.course.aggregate([

    $group: {
      _id: null,

    }
  },
  {
    $project: { _id: 0 }
  }
])
